\begin{frame}{UQ for agents --- setup}
\small
Estimate \textbf{trajectory success} from model uncertainty and tool feedback.

\vspace{0.5em}
\textbf{Models}
\begin{itemize}\setlength\itemsep{2pt}
  \item \texttt{gpt-oss-20b}
  \item \texttt{Qwen/Qwen3.6-35B-A3B} (VLM)
\end{itemize}

\vspace{0.3em}
\textbf{Agents}
\begin{itemize}\setlength\itemsep{2pt}
  \item LangGraph ReAct, smolagents
  \item Specialized agents (e.g., OSWorld and WebArena Verified)
  \item mini-swe-agent (DeepSWE)
\end{itemize}

\vspace{0.3em}
\textbf{Environments}\\[3pt]
Coding, interactive planning, real-world tool use,\\
web search, math, computer use.
\end{frame}

\begin{frame}{Bayesian belief state + continuous UQ}
\small
\textbf{Belief:} $b=P(Y=1\mid\text{trajectory evidence})$, where $Y=1$ means success.

\begin{block}{1. Tool / critic evidence}
\vspace{-0.3em}
\[
 \operatorname{logit}b_{\rm crit}
 =\operatorname{logit}b_0+
 \frac{1}{N}\sum_{t=1}^{N}\sum_i\log\frac{P(z_{t,i}\mid Y=1)}{P(z_{t,i}\mid Y=0)}.
\]
\vspace{-0.6em}
\footnotesize
$z_{t,i}$: critic $i$ at step $t$ (format valid, action admissible, no repeat,
tool success, state changed). \textbf{Tempered}: evidence averaged over the $N$ steps,
not multiplied --- one likelihood ratio per step saturates after a few steps and the
ranking degenerates into a count of bad steps (i.e.\ length).
\par\vspace{0.2em}
\end{block}

\begin{block}{2. Continuous UQ evidence}
\small
Fit score distributions $s\mid Y=y\sim\mathcal N(\mu_y,\sigma_y^2)$.
\vspace{-0.3em}
\[
 \operatorname{logit}b_{\rm fused}
 =\operatorname{logit}b_{\rm crit}
 +\lambda\sum_j\log
 \frac{\mathcal N(s_j;\mu_1,\sigma_1^2)}
      {\mathcal N(s_j;\mu_0,\sigma_0^2)}.
\]
\vspace{-0.6em}
\footnotesize
$s$: UQ score; $j$: generation index. SEP: binarised $s_j$ at a fitted threshold.
\par\vspace{0.2em}
\end{block}

\footnotesize
\textbf{UQ-only:} start from $b_0$.\quad
$b_0$: prior. All parameters fitted on the calibration half only.
\end{frame}
